# Canvas 2.0 MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Canvas 2.0 — session-aware tabbed workspace with live HTML widgets (Email, Todos, Contacts, Files, Notes) sharing the agent's data layer, replacing the current Workspace panel.

**Architecture:** Backend serves static HTML widgets + new WS event types (switch_tab, widget_refresh, new_session). Flutter renders workspace panel as WebView tab container + chat panel with per-session state. CoreMem `session_id` is wired through for per-session conversation isolation.

**Tech Stack:** Python 3.13 (FastAPI), Flutter (Riverpod, webview_flutter), CoreMem 0.9.1, HybridDB 0.4.5

```python
def get_messages(
    self, start_date: date | None = None, end_date: date | None = None,
    limit: int | None = None,
    session_id: str | None = None,
) -> list[Message]:
    ts_after = f"{start_date.isoformat()}T00:00:00" if start_date else None
    ts_before = f"{end_date.isoformat()}T23:59:59" if end_date else None
    kwargs: dict[str, Any] = {}
    if session_id is not None:
        kwargs["session_id"] = session_id
    memories = self._core.fetch(
        limit=limit or 10000,
        ts_after=ts_after,
        ts_before=ts_before,
        **kwargs,
    )
    return [self._to_msg(m) for m in reversed(memories)]
```

```python
import re

def parse_and_apply_directives(text: str, connection_state: dict) -> tuple[str, list[dict]]:
    """Parse directives from agent text, apply state changes, return (cleaned_text, events)."""
    events = []
    
    # [SWITCH_TAB:<tab_name>]
    switch_match = re.search(r'\[SWITCH_TAB:(\w+)\]', text)
    if switch_match:
        tab = switch_match.group(1).lower()
        connection_state["current_tab"] = tab
        connection_state["current_selected_item"] = None
        events.append({"type": WS_EVENT_SWITCH_TAB, "tab": tab})
        text = text.replace(switch_match.group(0), "").strip()
    
    # [WIDGET_REFRESH:<tab_name>]
    refresh_match = re.search(r'\[WIDGET_REFRESH:(\w+)\]', text)
    if refresh_match:
        tab = refresh_match.group(1).lower()
        events.append({"type": WS_EVENT_WIDGET_REFRESH, "tab": tab})
        text = text.replace(refresh_match.group(0), "").strip()
    
    # [WIDGET_REFRESH] (all)
    all_refresh = re.search(r'\[WIDGET_REFRESH\]', text)
    if all_refresh:
        events.append({"type": WS_EVENT_WIDGET_REFRESH, "tab": "all"})
        text = text.replace(all_refresh.group(0), "").strip()
    
    # [NEW_SESSION]
    new_session = re.search(r'\[NEW_SESSION\]', text)
    if new_session:
        events.append({"type": WS_EVENT_NEW_SESSION})
        text = text.replace(new_session.group(0), "").strip()
    
    return text, events
```

```python
"""Widget HTML templates for Canvas 2.0."""

from pathlib import Path
from fastapi import APIRouter, HTTPException
from fastapi.responses import HTMLResponse

router = APIRouter()

WIDGETS_DIR = Path(__file__).parent.parent / "static" / "widgets"

AVAILABLE_WIDGETS = {
    "email": "email.html",
    "todos": "todos.html",
    "contacts": "contacts.html",
    "files": "files.html",
}


@router.get("/widget/{widget_type}", response_class=HTMLResponse)
async def get_widget(widget_type: str):
    widget_type = widget_type.lower()
    if widget_type not in AVAILABLE_WIDGETS:
        raise HTTPException(status_code=404, detail=f"Widget '{widget_type}' not found")
    
    file_path = WIDGETS_DIR / AVAILABLE_WIDGETS[widget_type]
    if not file_path.exists():
        raise HTTPException(status_code=404, detail=f"Widget template not found")
    
    html = file_path.read_text(encoding="utf-8")
    return HTMLResponse(content=html)
```

```dart
class ChatTabNotifier extends StateNotifier<ChatTabState> {
  ChatTabNotifier() : super(ChatTabState(sessions: {'default': 'Chat 1'}, activeSessionId: 'default'));
  
  void openSession(String id, String label) {
    state = state.copyWith(sessions: {...state.sessions, id: label}, activeSessionId: id);
  }
  
  void closeSession(String id) {
    final updated = Map<String, String>.from(state.sessions)..remove(id);
    final nextActive = state.activeSessionId == id
        ? updated.keys.firstOrNull ?? 'default'
        : state.activeSessionId;
    state = state.copyWith(sessions: updated, activeSessionId: nextActive);
  }
  
  void switchToSession(String id) {
    state = state.copyWith(activeSessionId: id);
  }
}
```
